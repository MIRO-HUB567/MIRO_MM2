local function trim(s) return s:match("^%s*(.-)%s*$") end
local function fetchHTML(url)
    local res = request({Url = url, Method = "GET", Headers = headers})
    return res and res.Body or ""
end
local function parseValue(div)
    local s = div:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if s then return tonumber(s:gsub(",", "")) end
    return nil
end
local function extractItems(html)
    local t = {}
    for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        name = name:match("([^<]+)")
        if name then
            name = trim(name:gsub("%s+", " ")):lower()
            t[name] = parseValue(body)
        end
    end
    return t
end
local function buildValueList()
    local godlies, ancients, uniques, vintages, chromas = {}, {}, {}, {}, {}
    local function fetch(url, dest)
        task.spawn(function()
            local html = fetchHTML(url)
            if html then
                for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
                    name = name:match("([^<]+)")
                    if name then
                        name = trim(name:gsub("%s+", " ")):lower()
                        dest[name] = parseValue(body)
                    end
                end
            end
        end)
    end
    fetch(categories.godly, godlies)
    fetch(categories.ancient, ancients)
    fetch(categories.unique, uniques)
    fetch(categories.classic, vintages)
    fetch(categories.chroma, chromas)
    task.wait(2)
    local valueList = {}
    for id, item in pairs(database) do
        local name = item.ItemName and item.ItemName:lower() or ""
        local rarity = item.Rarity or ""
        local isChroma = item.Chroma or false
        if name ~= "" and rarity ~= "" then
            local rIndex = table.find(rarityTable, rarity)
            local gIndex = table.find(rarityTable, "Godly")
            if rIndex and gIndex and rIndex >= gIndex then
                local val
                if isChroma then
                    for cName, cVal in pairs(chromas) do
                        if name:find(cName) then val = cVal break end
                    end
                else
                    val = godlies[name] or ancients[name] or uniques[name] or vintages[name]
                end
                if val then valueList[id] = val end
            end
        end
    end
    return valueList
end
